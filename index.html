function doPost(e) { try { const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); const sheet = spreadsheet.getSheets()[0]; const data = e.parameter; if (!data.beer || !data.brewery) { throw new Error("Beer and brewery are required."); } let imageUrl = ""; // Optional beer photo: JPG/JPEG or PNG, maximum 5 MB. if (data.imageData) { const allowedTypes = ["image/jpeg", "image/png"]; if (!allowedTypes.includes(String(data.imageType || "").toLowerCase())) { throw new Error("Only JPG and PNG images are allowed."); } // Base64 data URLs are sent by the website. const match = String(data.imageData).match(/^data:(image\/(?:jpeg|png));base64,(.+)$/i); if (!match) { throw new Error("The uploaded image could not be read."); } const base64 = match[2]; const estimatedBytes = Math.floor((base64.length * 3) / 4); if (estimatedBytes > 5 * 1024 * 1024) { throw new Error("The uploaded image is larger than 5 MB."); } // Find the existing Google Drive folder named exactly: // Beer website images const folders = DriveApp.getFoldersByName("Beer website images"); if (!folders.hasNext()) { throw new Error('Google Drive folder "Beer website images" was not found.'); } const folder = folders.next(); const bytes = Utilities.base64Decode(base64); const safeOriginalName = String(data.imageName || "beer-photo") .replace(/[^a-zA-Z0-9._-]/g, "_"); const filename = new Date().getTime() + "_" + safeOriginalName; const blob = Utilities.newBlob( bytes, String(data.imageType).toLowerCase(), filename ); const file = folder.createFile(blob); // The image must be viewable by visitors to the public website. file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); imageUrl = "https://drive.google.com/uc?export=view&id=" + file.getId(); } // Columns: // Date | Beer | Brewery | Pub / Venue | County | Rating | Tasting Note | Approved | Image URL sheet.appendRow([ new Date(), data.beer || "", data.brewery || "", data.pub || "", data.county || "", data.rating || "", data.note || "", "NO", imageUrl ]); return ContentService .createTextOutput("OK") .setMimeType(ContentService.MimeType.TEXT); } catch (error) { return ContentService .createTextOutput("ERROR: " + error.message) .setMimeType(ContentService.MimeType.TEXT); } } function doGet(e) { try { const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); const sheet = spreadsheet.getSheets()[0]; const values = sheet.getDataRange().getDisplayValues(); if (values.length <= 1) { return ContentService .createTextOutput("[]") .setMimeType(ContentService.MimeType.JSON); } values.shift(); const approved = values .filter(row => String(row[7]).toUpperCase() === "YES") .map(row => ({ date: row[0], beer: row[1], brewery: row[2], pub: row[3], county: row[4], rating: row[5], note: row[6], imageUrl: row[8] || "" })); return ContentService .createTextOutput(JSON.stringify(approved)) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService .createTextOutput(JSON.stringify({ success: false, message: error.message })) .setMimeType(ContentService.MimeType.JSON); } }
